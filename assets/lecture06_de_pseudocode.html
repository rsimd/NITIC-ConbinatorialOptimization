\begin{algorithm}
\caption{DE/rand/1/bin}
\begin{algorithmic}
\REQUIRE objective function $f$, bounds, population size $NP$, scale factor $F$, crossover rate $CR$, maximum generations $T_{\max}$
\STATE initialize population $X=\{x_1,\ldots,x_{NP}\}$ uniformly within bounds
\STATE evaluate $f(x_i)$ for all individuals
\FOR{$t = 1,\ldots,T_{\max}$}
  \FOR{each target individual $x_i$}
    \STATE choose distinct indices $r_1,r_2,r_3$ different from $i$
    \STATE $v_i \gets x_{r_1} + F(x_{r_2}-x_{r_3})$
    \STATE clip $v_i$ to bounds if needed
    \STATE choose forced dimension $j_{\mathrm{rand}}$
    \FOR{each dimension $j$}
      \IF{$\mathrm{rand}_j \le CR$ or $j=j_{\mathrm{rand}}$}
        \STATE $u_{i,j} \gets v_{i,j}$
      \ELSE
        \STATE $u_{i,j} \gets x_{i,j}$
      \ENDIF
    \ENDFOR
    \IF{$f(u_i) \le f(x_i)$}
      \STATE $x_i \gets u_i$
    \ENDIF
  \ENDFOR
\ENDFOR
\RETURN best individual in $X$
\end{algorithmic}
\end{algorithm}
