\begin{algorithm}
\caption{GA-TSP main loop}
\begin{algorithmic}
\REQUIRE city coordinates, population size $M$, max generations $T$, crossover rate $p_c$, mutation rate $p_m$
\ENSURE best tour $\pi^*$
\STATE generate initial population $P(0)$ of permutations
\STATE evaluate each individual and compute fitness $g_i$
\FOR{$t = 0,\ldots,T-1$}
  \STATE $P' \gets$ select parents from $P(t)$ using fitness values $g_i$
  \STATE $P'' \gets$ apply crossover to $P'$ with probability $p_c$
  \STATE $P(t+1) \gets$ apply mutation to $P''$ with probability $p_m$ and keep elite individuals
  \STATE evaluate each individual and update fitness values $g_i$
\ENDFOR
\RETURN best tour $\pi^*$ found so far
\end{algorithmic}
\end{algorithm}
