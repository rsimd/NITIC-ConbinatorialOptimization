\begin{algorithm}
\caption{Ant System for symmetric TSP}
\begin{algorithmic}
\REQUIRE symmetric distance matrix $D$, number of ants $M$, iterations $K$
\REQUIRE parameters $\alpha > 0$, $\beta > 0$, $\rho \in (0,1)$, $Q > 0$
\ENSURE best tour $\pi^*$ and its length $L^*$
\STATE initialize pheromone matrix $\tau$ with small positive values
\STATE $L^* \gets +\infty$
\FOR{$t = 1,\ldots,K$}
  \STATE $S \gets \emptyset$
  \FOR{$k = 1,\ldots,M$}
    \STATE choose a start city
    \STATE construct tour $\pi$ by sampling next cities according to Eq. (2)
    \STATE compute tour length $L$
    \STATE $S \gets S \cup \{(\pi,L)\}$
    \IF{$L < L^*$}
      \STATE $\pi^* \gets \pi$, $L^* \gets L$
    \ENDIF
  \ENDFOR
  \STATE evaporate all pheromones according to Eq. (3)
  \STATE deposit pheromones for tours in $S$ according to Eqs. (4)(5)
\ENDFOR
\RETURN $\pi^*, L^*$
\end{algorithmic}
\end{algorithm}
