\begin{algorithm}
\caption{CMA-ES (minimization)}
\begin{algorithmic}
\REQUIRE objective function $f$, initial mean $m$, initial step size $\sigma$, maximum generations $T$
\STATE $C \gets I$, $p_\sigma \gets 0$, $p_c \gets 0$
\STATE choose $\lambda, \mu$, weights $w_i$, and learning rates from dimension $n$
\FOR{$t = 1,\ldots,T$}
  \STATE decompose $C = B D^2 B^\top$
  \FOR{$k = 1,\ldots,\lambda$}
    \STATE sample $z_k \sim \mathcal{N}(0, I)$
    \STATE $y_k \gets B D z_k$
    \STATE $x_k \gets m + \sigma y_k$
    \STATE $f_k \gets f(x_k)$
  \ENDFOR
  \STATE sort candidates by $f_k$ and select the best $\mu$
  \STATE $y_w \gets \sum_{i=1}^{\mu} w_i y_{i:\lambda}$
  \STATE $m \gets m + \sigma y_w$
  \STATE update $p_\sigma$ and $\sigma$
  \STATE update $p_c$ and $C$ by rank-one and rank-$\mu$ updates
\ENDFOR
\RETURN best-so-far $x$
\end{algorithmic}
\end{algorithm}
