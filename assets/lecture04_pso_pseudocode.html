\begin{algorithm}
\caption{PSO (minimization)}
\begin{algorithmic}
\REQUIRE objective function $f$, swarm size $N$, dimension $d$, max iterations $T_{\max}$, bounds $[x_{\min},x_{\max}]^d$
\REQUIRE parameters $w$, $c_1$, $c_2$, and optional velocity limit $v_{\max}$
\ENSURE global best position $g$
\FOR{$i = 1,\ldots,N$}
  \STATE initialize $x_i$ uniformly in the domain
  \STATE initialize $v_i$ as a small random vector or zero vector
  \STATE $p_i \gets x_i$
\ENDFOR
\STATE $g \gets \arg\min_{x \in \{p_1,\ldots,p_N\}} f(x)$
\FOR{$t = 1,\ldots,T_{\max}$}
  \FOR{$i = 1,\ldots,N$}
    \FOR{$k = 1,\ldots,d$}
      \STATE draw $r_1,r_2 \sim U(0,1)$
      \STATE $v_{i,k} \gets w v_{i,k} + c_1 r_1(p_{i,k}-x_{i,k}) + c_2 r_2(g_k-x_{i,k})$
      \STATE clip $v_{i,k}$ to $[-v_{\max},v_{\max}]$ if needed
    \ENDFOR
    \STATE $x_i \gets x_i + v_i$
    \STATE project $x_i$ into the feasible domain if needed
    \IF{$f(x_i) < f(p_i)$}
      \STATE $p_i \gets x_i$
    \ENDIF
    \IF{$f(x_i) < f(g)$}
      \STATE $g \gets x_i$
    \ENDIF
  \ENDFOR
\ENDFOR
\RETURN $g$
\end{algorithmic}
\end{algorithm}
